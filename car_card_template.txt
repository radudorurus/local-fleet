# This layout card organizes vehicle data into a two-column grid structure.
# The picture block displays the vehicle identification image.
# The car info card renders the vehicle model and VIN string.
# The mileage card tracks current distance values in kilometers.
# The service km card calculates distance to the next maintenance.
# Dynamic color coding highlights maintenance urgency thresholds.
# The service days card monitors remaining time until inspection.
# The ITP card tracks mandatory technical inspection deadlines.
# The RCA card tracks compulsory insurance validity periods.
# The vignette card manages road tax expiration dates.
# The tire type card displays the active seasonal tire configuration.
# It warns if the installed tire type differs from the recommended set.
# The tire age card computes the maximum age of the active tire set.
# It appends a replacement notice if the age exceeds eight years.
# Icons and colors dynamically change based on specific age thresholds.
# For the recommendation to change the tire set based on season/temperature there is a helper sensor 'input_select.recommended_tire', a historic helper 'sensor.garage_ht_sensor_outside_low_temp_days' (temperature entity, state= <7, type of sensor= time, Start = empty, End= {{now()}}, duration= 3 days, class= Measurement) based on the temperature from 'sensor.ht_sensor_outside_temperature' and an automation which sets this sensor based on the outside history temperature and the weather prognosis for 5 days in advance.
# You need to create the helper using your external temperature sensor.
# and replace in the automation the historic helper name and the external temperature sensor name
# If this is too complicated you may adjust the card code to ignore 'input_select.recommended_tire' sensor
################################@
# the automation code
#################################
# alias: recommended_tire
# description: ''
# triggers:
#   - trigger: time
#     at: '06:00:00'
#     weekday:
#       - sun
#       - mon
#       - tue
#       - sat
#       - fri
#       - thu
#       - wed
# conditions: []
# actions:
#   - action: weather.get_forecasts
#     metadata: {}
#     target:
#       entity_id: weather.forecast_home
#     data:
#       type: daily
#     response_variable: prognoza_meteo
#   - choose:
#       - conditions:
#           - condition: template
#             value_template: >
#               {% set forecast = prognoza_meteo['weather.forecast_home'].forecast
#               %} {% set ns = namespace(cold_days=0) %} {% for day in
#               forecast[:5] %}
#                 {% if day.templow is defined and day.templow < 7 %}
#                   {% set ns.cold_days = ns.cold_days + 1 %}
#                 {% endif %}
#               {% endfor %} {{ (ns.cold_days >= 3) or
#               (states('sensor.garage_ht_sensor_outside_low_temp_days') |
#               float(0) >= 24) or (states('sensor.ht_sensor_outside_temperature')
#               | float(10) < 7) }}
#         sequence:
#           - target:
#               entity_id: input_select.recommended_tire
#             data:
#               option: Winter
#             action: input_select.select_option
#       - conditions:
#           - condition: template
#             value_template: >
#               {% set forecast = prognoza_meteo['weather.forecast_home'].forecast
#               %} {% set ns = namespace(warm_days=0) %} {% for day in
#               forecast[:5] %}
#                 {% if day.templow is defined and day.templow >= 10 %}
#                   {% set ns.warm_days = ns.warm_days + 1 %}
#                 {% endif %}
#               {% endfor %} {{ (ns.warm_days >= 4) and
#               (states('sensor.ht_sensor_outside_temperature') | float(0) >= 10)
#               }}
#         sequence:
#           - target:
#               entity_id: input_select.recommended_tire
#             data:
#               option: Summer
#             action: input_select.select_option
